cmake_minimum_required(VERSION 3.20)
project(MXLoggerJavaApiTests LANGUAGES C CXX)
enable_testing()
get_filename_component(MXLOGGER_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../.." ABSOLUTE)
