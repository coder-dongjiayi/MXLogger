cmake_minimum_required(VERSION 3.19)
project(flutter_mxlogger LANGUAGES NONE)

# ffiPlugin: true. Only deploy precompiled binaries; never compile/download Core
# inside a consuming application. Profile uses the Release binary.
if(DEFINED FLUTTER_TARGET_PLATFORM AND NOT FLUTTER_TARGET_PLATFORM STREQUAL "windows-x64")
    message(FATAL_ERROR "flutter_mxlogger currently ships Windows x64 DLLs only")
endif()
if(DEFINED CMAKE_SIZEOF_VOID_P AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "flutter_mxlogger requires a 64-bit Windows application")
endif()
set(MXLOGGER_BINARY_ROOT "${CMAKE_CURRENT_LIST_DIR}/bin/x64")
if(NOT "$ENV{MXLOGGER_WINDOWS_BINARY_DIR}" STREQUAL "")
    set(MXLOGGER_BINARY_ROOT "$ENV{MXLOGGER_WINDOWS_BINARY_DIR}")
endif()
if(NOT EXISTS "${MXLOGGER_BINARY_ROOT}/manifest.json")
    message(FATAL_ERROR "Missing precompiled MXLogger DLLs. Restore the local MXLogger.Native NuGet package and run scripts/prepare_flutter_windows.py --package <nupkg>. See README_CN.md. No Windows Core sources are bundled.")
endif()
file(READ "${MXLOGGER_BINARY_ROOT}/manifest.json" MXLOGGER_MANIFEST)
string(JSON MXLOGGER_BINARY_VERSION GET "${MXLOGGER_MANIFEST}" core_version)
string(JSON MXLOGGER_BINARY_ARCH GET "${MXLOGGER_MANIFEST}" architecture)
string(JSON MXLOGGER_BINARY_ABI GET "${MXLOGGER_MANIFEST}" ffi_abi)
if(NOT MXLOGGER_BINARY_VERSION STREQUAL "2.3.0" OR
   NOT MXLOGGER_BINARY_ARCH STREQUAL "x64" OR NOT MXLOGGER_BINARY_ABI EQUAL 1)
    message(FATAL_ERROR "Incompatible MXLogger Windows DLL version, architecture or FFI ABI")
endif()
foreach(configuration Debug Release)
    set(dll "${MXLOGGER_BINARY_ROOT}/${configuration}/mxlogger_ffi.dll")
    if(NOT EXISTS "${dll}")
        message(FATAL_ERROR "Missing ${configuration} DLL: ${dll}")
    endif()
    string(JSON expected_hash GET "${MXLOGGER_MANIFEST}" sha256 "${configuration}")
    file(SHA256 "${dll}" actual_hash)
    if(NOT actual_hash STREQUAL expected_hash)
        message(FATAL_ERROR "MXLogger ${configuration} DLL hash mismatch; restore the matching package again")
    endif()
endforeach()
if(NOT CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    set(flutter_mxlogger_bundled_libraries
        "${MXLOGGER_BINARY_ROOT}/$<IF:$<CONFIG:Debug>,Debug,Release>/mxlogger_ffi.dll"
        PARENT_SCOPE)
endif()
