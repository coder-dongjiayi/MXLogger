cmake_minimum_required(VERSION 3.14)
project(MXLoggerOHOS LANGUAGES C CXX)
get_filename_component(MXLOGGER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../.." ABSOLUTE)
add_subdirectory("${MXLOGGER_ROOT}/Core" "${CMAKE_CURRENT_BINARY_DIR}/core")
target_compile_definitions(MXLoggerCore PRIVATE MXLOGGER_CONSOLE_ENABLED=0 MXLOGGER_DEBUG_LOG_ENABLED=0)
set_target_properties(MXLoggerCore PROPERTIES CXX_VISIBILITY_PRESET hidden
    C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_library(mxlogger SHARED ffi_bridge.cpp)
target_include_directories(mxlogger PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${MXLOGGER_ROOT}/Core/ffi")
target_compile_definitions(mxlogger PRIVATE MXLOGGER_FFI_BUILD=1)
target_compile_features(mxlogger PUBLIC cxx_std_17)
target_link_libraries(mxlogger PRIVATE MXLoggerCore)
set_target_properties(mxlogger PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(OHOS)
    target_sources(mxlogger PRIVATE napi_bridge.cpp)
    target_link_libraries(mxlogger PRIVATE libace_napi.z.so libhilog_ndk.z.so)
endif()
option(MXLOGGER_OHOS_BUILD_TESTS "Build managed-bridge contract and concurrency tests" OFF)
if(MXLOGGER_OHOS_BUILD_TESTS)
    enable_testing()
    add_executable(mxlogger_ohos_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tests/bridge_test.cpp")
    target_link_libraries(mxlogger_ohos_tests PRIVATE mxlogger)
    add_test(NAME MXLoggerOHOSBridge COMMAND mxlogger_ohos_tests)
endif()
