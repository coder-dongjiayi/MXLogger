cmake_minimum_required(VERSION 3.20)
# Core has its own older minimum; select the same CRT policy in both projects.
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(MXLoggerDemo LANGUAGES C CXX)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory(../../Core core)
add_library(mxlogger_demo_model STATIC demo_model.cpp demo_process.cpp demo_smoke.cpp)
target_compile_features(mxlogger_demo_model PUBLIC cxx_std_17)
target_include_directories(mxlogger_demo_model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mxlogger_demo_model PUBLIC MXLoggerCore)

if(WIN32)
    enable_language(RC)
    set(DEMO_LOGO "${CMAKE_CURRENT_SOURCE_DIR}/../../iOS/MXLoggerDemo/MXLoggerDemo/Assets.xcassets/mxlogger_logo.imageset/mxlogger_logo.png")
    set(DEMO_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/demo.manifest")
    configure_file(demo.rc.in "${CMAKE_CURRENT_BINARY_DIR}/demo.rc" @ONLY)
    add_executable(MXLoggerDemo WIN32 main_win.cpp "${CMAKE_CURRENT_BINARY_DIR}/demo.rc")
    target_compile_definitions(mxlogger_demo_model PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(MXLoggerDemo PRIVATE mxlogger_demo_model comctl32 gdiplus ole32 shell32 uuid uxtheme)
    if(MSVC)
        target_compile_options(MXLoggerDemo PRIVATE /utf-8 /W4)
        target_compile_options(mxlogger_demo_model PRIVATE /W4)
        # The resource contains the complete DPI/common-controls manifest.
        target_link_options(MXLoggerDemo PRIVATE /MANIFEST:NO)
    endif()
    install(TARGETS MXLoggerDemo RUNTIME DESTINATION .)
    install(FILES README.md ../../LICENSE.TXT DESTINATION .)
elseif(APPLE)
    # Runs the exact GUI action model on a Mac; it does not emulate the Win32 UI.
    add_executable(MXLoggerDemoModelCheck main_model.cpp)
    target_link_libraries(MXLoggerDemoModelCheck PRIVATE mxlogger_demo_model)
    target_compile_options(mxlogger_demo_model PRIVATE -Wall -Wextra -Wpedantic)
    enable_testing()
    add_test(NAME windows-demo-model COMMAND MXLoggerDemoModelCheck --self-test
             --report-dir "${CMAKE_BINARY_DIR}/self-test")
    set_tests_properties(windows-demo-model PROPERTIES TIMEOUT 120)
else()
    message(FATAL_ERROR "Build the Windows UI on Windows, or its action model on macOS.")
endif()
